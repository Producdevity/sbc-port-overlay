cmake_minimum_required(VERSION 3.20)
project(sbc_port_overlay VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
option(SPO_BUILD_TESTS "Build host tests" ON)
option(SPO_BUILD_GL_TESTS "Build EGL/GLES integration tests (Linux Mesa)" OFF)
option(SPO_BUILD_SDL2 "Build optional SDL2 adapter" OFF)
option(SPO_BUILD_EXAMPLES "Build integration examples and SDL demo" OFF)
option(SPO_SDL_LINK_SHARED "Link the SDL adapter to the shared variant" OFF)
set(SPO_SOURCES src/core.c src/input.c src/config.c src/metrics.c src/sampler.c src/layout.c src/font.c src/gles.c)
add_library(spo_objects OBJECT ${SPO_SOURCES})
target_compile_features(spo_objects PUBLIC c_std_11)
target_compile_definitions(spo_objects PRIVATE _POSIX_C_SOURCE=200809L)
target_include_directories(spo_objects PRIVATE include)
target_link_libraries(spo_objects PRIVATE Threads::Threads)
set_target_properties(spo_objects PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(spo_objects PRIVATE -Wall -Wextra -Wpedantic)
endif()
foreach(variant static shared)
  string(TOUPPER ${variant} kind)
  add_library(spo_${variant} ${kind} $<TARGET_OBJECTS:spo_objects>)
  add_library(spo::${variant} ALIAS spo_${variant})
  target_include_directories(spo_${variant} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_link_libraries(spo_${variant} PRIVATE m Threads::Threads)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(spo_${variant} PRIVATE pthread)
  endif()
  set_target_properties(spo_${variant} PROPERTIES OUTPUT_NAME spo EXPORT_NAME ${variant})
endforeach()
set_target_properties(spo_shared PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(SPO_BUILD_SDL2 OR SPO_BUILD_EXAMPLES)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET "sdl2>=2.0.6")
  add_library(spo_sdl2 STATIC src/sdl2.c)
  add_library(spo::sdl2 ALIAS spo_sdl2)
  set_target_properties(spo_sdl2 PROPERTIES EXPORT_NAME sdl2 POSITION_INDEPENDENT_CODE ON)
  if(SPO_SDL_LINK_SHARED)
    target_link_libraries(spo_sdl2 PUBLIC spo_shared PkgConfig::SDL2)
  else()
    target_link_libraries(spo_sdl2 PUBLIC spo_static PkgConfig::SDL2)
  endif()
endif()
if(SPO_BUILD_EXAMPLES)
  enable_language(CXX)
  add_executable(spo_demo examples/demo.c)
  target_link_libraries(spo_demo PRIVATE spo_sdl2 ${CMAKE_DL_LIBS})
  target_compile_definitions(spo_demo PRIVATE _POSIX_C_SOURCE=200809L)
  add_executable(spo_c examples/c/main.c)
  target_link_libraries(spo_c PRIVATE spo_static)
  add_executable(spo_cpp examples/cpp/main.cpp)
  target_compile_features(spo_cpp PRIVATE cxx_std_11)
  target_link_libraries(spo_cpp PRIVATE spo_static)
endif()
if(SPO_BUILD_TESTS)
  enable_testing()
  add_executable(spo_tests tests/test_core.c)
  target_include_directories(spo_tests PRIVATE src)
  target_compile_definitions(spo_tests PRIVATE _POSIX_C_SOURCE=200809L SPO_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  target_link_libraries(spo_tests PRIVATE spo_static m)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(spo_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME core COMMAND spo_tests)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(spo_sampler_tests tests/test_sampler.c)
    target_link_libraries(spo_sampler_tests PRIVATE spo_static Threads::Threads)
    target_compile_definitions(spo_sampler_tests PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(spo_sampler_tests PRIVATE -UNDEBUG)
    target_link_options(spo_sampler_tests PRIVATE -Wl,--wrap=pread -Wl,--wrap=pthread_mutex_lock)
    add_test(NAME sampler COMMAND spo_sampler_tests)
    set_tests_properties(sampler PROPERTIES TIMEOUT 15)
  endif()
  if(TARGET spo_sdl2)
    add_executable(spo_sdl_tests tests/test_sdl.c)
    target_link_libraries(spo_sdl_tests PRIVATE spo_sdl2)
    target_compile_options(spo_sdl_tests PRIVATE -UNDEBUG)
    add_test(NAME sdl_adapter COMMAND spo_sdl_tests)
  endif()
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME tooling COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.py)
  endif()
  if(SPO_BUILD_GL_TESTS)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
    pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
    add_executable(spo_gl_tests tests/test_gl.c)
    target_compile_options(spo_gl_tests PRIVATE -UNDEBUG)
    target_link_libraries(spo_gl_tests PRIVATE spo_static PkgConfig::EGL PkgConfig::GLES)
    add_test(NAME gles2 COMMAND spo_gl_tests 2)
    add_test(NAME gles2_no_vao COMMAND spo_gl_tests 2 no-vao)
    add_test(NAME gles3 COMMAND spo_gl_tests 3)
    set_tests_properties(gles2 gles2_no_vao PROPERTIES ENVIRONMENT "EGL_PLATFORM=surfaceless;MESA_GLES_VERSION_OVERRIDE=2.0" SKIP_RETURN_CODE 77)
    set_tests_properties(gles3 PROPERTIES ENVIRONMENT "EGL_PLATFORM=surfaceless;MESA_GLES_VERSION_OVERRIDE=3.2" SKIP_RETURN_CODE 77)
  endif()
endif()
install(TARGETS spo_static spo_shared EXPORT spoTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(TARGET spo_sdl2)
  install(TARGETS spo_sdl2 EXPORT spoSdlTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(EXPORT spoSdlTargets NAMESPACE spo:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spo)
endif()
install(FILES include/spo.h include/spo.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/sbc-port-overlay)
install(FILES vendor/spleen/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/sbc-port-overlay RENAME FONT-LICENSE)
install(EXPORT spoTargets NAMESPACE spo:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spo)
configure_package_config_file(cmake/spoConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/spoConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spo)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/spoConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/spoConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/spoConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/spo)
